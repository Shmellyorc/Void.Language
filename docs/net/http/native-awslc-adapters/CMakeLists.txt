cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
project(void_windows_tls_adapter_research LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "23.1.4")
  message(FATAL_ERROR "Use native Windows clang-cl 23.1.4")
endif()
get_filename_component(REPO_ROOT "../../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(TLS_PREFIX "${REPO_ROOT}/.toolchains/native-awslc/${CMAKE_BUILD_TYPE}")
set(AWS_SOURCE "${REPO_ROOT}/.toolchains/native-awslc/source/aws-lc-cd481c207e38f06aeb5766f4a1cdfce10a4c0af8")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(CURL_ARCHIVE "${TLS_PREFIX}/curl/curl/lib/libcurl-d.lib")
else()
  set(CURL_ARCHIVE "${TLS_PREFIX}/curl/curl/lib/libcurl.lib")
endif()
add_executable(adapter-probe probe.cpp)
add_executable(fixture fixture.cpp)
add_executable(root-audit roots.cpp)
foreach(_target adapter-probe fixture root-audit)
  target_compile_features(${_target} PRIVATE cxx_std_17)
  target_compile_options(${_target} PRIVATE /W4 /EHsc)
  target_compile_definitions(${_target} PRIVATE CURL_STATICLIB)
  target_include_directories(${_target} PRIVATE "${REPO_ROOT}/third_party/curl/include" "${AWS_SOURCE}/include")
  target_link_libraries(${_target} PRIVATE "${CURL_ARCHIVE}" "${TLS_PREFIX}/aws-lc/ssl/ssl.lib"
    "${TLS_PREFIX}/aws-lc/crypto/crypto.lib" "${TLS_PREFIX}/prefix/lib/nghttp2.lib"
    normaliz bcrypt advapi32 crypt32 ncrypt secur32 ws2_32 iphlpapi)
endforeach()
