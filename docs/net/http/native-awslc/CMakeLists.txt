cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
project(void_windows_awslc_research LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "23.1.4")
  message(FATAL_ERROR "Use native Windows clang-cl 23.1.4")
endif()
option(STUDY_MULTI "Compile Schannel alongside AWS-LC for selection research" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
foreach(_protocol DICT FILE FTP GOPHER IMAP LDAP LDAPS MQTT POP3 RTSP SMTP TELNET TFTP WEBSOCKETS IPFS)
  set(CURL_DISABLE_${_protocol} ON CACHE BOOL "" FORCE)
endforeach()
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ${STUDY_MULTI} CACHE BOOL "" FORCE)
set(CURL_WINDOWS_SSPI ON CACHE BOOL "" FORCE)
set(ENABLE_UNICODE ON CACHE BOOL "" FORCE)
set(USE_WIN32_IDN ON CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS ON)
set(OPENSSL_INCLUDE_DIR "${AWS_SOURCE}/include")
set(OPENSSL_SSL_LIBRARY "${AWS_BUILD}/ssl/ssl.lib")
set(OPENSSL_CRYPTO_LIBRARY "${AWS_BUILD}/crypto/crypto.lib")
set(OPENSSL_MSVC_STATIC_RT ON)
# CMake's Windows FindOpenSSL selects these configuration-specific variables.
foreach(_configuration DEBUG RELEASE)
  set(LIB_EAY_${_configuration} "${AWS_BUILD}/crypto/crypto.lib" CACHE FILEPATH "Pinned AWS-LC" FORCE)
  set(SSL_EAY_${_configuration} "${AWS_BUILD}/ssl/ssl.lib" CACHE FILEPATH "Pinned AWS-LC" FORCE)
endforeach()
# clang-cl uses the Windows C++ runtime, not libstdc++.
set(_curl_stdlib_cxx "")
set(USE_NGHTTP2 ON CACHE BOOL "" FORCE)
set(NGHTTP2_USE_STATIC_LIBS ON)
set(NGHTTP2_INCLUDE_DIR "${H2_PREFIX}/include")
set(NGHTTP2_LIBRARY "${H2_PREFIX}/lib/nghttp2.lib")
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(ENABLE_ARES OFF CACHE BOOL "" FORCE)
set(CURL_CA_BUNDLE none CACHE STRING "No ambient file fallback in the probe" FORCE)
set(CURL_CA_PATH none CACHE STRING "No ambient directory fallback in the probe" FORCE)
set(CURL_CA_FALLBACK OFF CACHE BOOL "" FORCE)
set(CURL_DISABLE_OPENSSL_AUTO_LOAD_CONFIG ON CACHE BOOL "" FORCE)
find_package(OpenSSL REQUIRED)
add_subdirectory(../../../../third_party/curl curl EXCLUDE_FROM_ALL)
add_executable(probe probe.cpp)
target_compile_features(probe PRIVATE cxx_std_17)
target_compile_options(probe PRIVATE /W4 /EHsc)
target_link_libraries(probe PRIVATE CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
