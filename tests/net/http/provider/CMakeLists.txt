cmake_minimum_required(VERSION 3.24)
project(VoidCurlRegression LANGUAGES C)
get_filename_component(VOID_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
include("${VOID_SOURCE_ROOT}/cmake/CurlProvider.cmake")
set(VOID_CURL_PROFILE COOKIE CACHE STRING "COOKIE, NO_COOKIES, NO_HTTP or HTTP")
void_add_curl_provider("${VOID_CURL_PROFILE}")
if(VOID_CURL_PROFILE STREQUAL "COOKIE" OR VOID_CURL_PROFILE STREQUAL "HTTP")
  set(probe cookie_accounting_probe)
else()
  set(probe cookie_disabled_probe)
endif()
add_executable(provider_probe "${CMAKE_CURRENT_SOURCE_DIR}/${probe}.c")
target_link_libraries(provider_probe PRIVATE CURL::libcurl_static)
if(MSVC)
  target_link_options(provider_probe PRIVATE /STACK:1048576)
endif()
if(NOT MSVC)
  target_compile_options(provider_probe PRIVATE -Wall -Wextra -Werror)
endif()
void_enable_sanitizers(provider_probe PRIVATE)
if(VOID_CURL_PROFILE STREQUAL "HTTP")
  enable_language(CXX)
  find_package(Threads REQUIRED)
  add_executable(url_probe "${VOID_SOURCE_ROOT}/tests/net/http/url.cpp"
    "${VOID_SOURCE_ROOT}/src/runtime/url.cpp" "${VOID_SOURCE_ROOT}/src/runtime/http_provider.cpp")
  target_compile_features(url_probe PRIVATE cxx_std_20)
  target_include_directories(url_probe PRIVATE "${VOID_SOURCE_ROOT}/include" "${VOID_HTTP_DEPS_ROOT}/include")
  target_link_libraries(url_probe PRIVATE CURL::libcurl_static Threads::Threads)
  void_enable_sanitizers(url_probe PRIVATE)
endif()
