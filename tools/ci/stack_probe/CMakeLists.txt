cmake_minimum_required(VERSION 3.24)
project(VoidStackProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Threads REQUIRED)
get_filename_component(VOID_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT PROBE_MODE MATCHES "^O[02]$")
  message(FATAL_ERROR "PROBE_MODE must be O0 or O2")
endif()
# Only the six LLVM-independent runtime files. No frontend, backend, ORC or tests.
add_library(probe_runtime STATIC
  "${VOID_ROOT}/src/runtime/heap.cpp" "${VOID_ROOT}/src/runtime/coordination.cpp"
  "${VOID_ROOT}/src/runtime/callback.cpp" "${VOID_ROOT}/src/runtime/abi.cpp"
  "${VOID_ROOT}/src/runtime/reload.cpp" "${VOID_ROOT}/src/runtime/native.cpp")
target_include_directories(probe_runtime PUBLIC "${VOID_ROOT}/include")
target_link_libraries(probe_runtime PUBLIC Threads::Threads)
add_library(probe_options INTERFACE)
if(MSVC)
  target_compile_definitions(probe_options INTERFACE _ITERATOR_DEBUG_LEVEL=0 _WIN32_WINNT=0x0A00)
  target_compile_options(probe_options INTERFACE /EHsc /W4
    "$<IF:$<STREQUAL:${PROBE_MODE},O0>,/Od,/O2>")
else()
  target_compile_options(probe_options INTERFACE -Wall -Wextra -Wpedantic "-${PROBE_MODE}")
endif()
target_link_libraries(probe_runtime PUBLIC probe_options)
foreach(kib IN ITEMS 64 128 256)
  set(object "${PROBE_OBJECT_DIR}/frame-${kib}.o")
  if(NOT EXISTS "${object}")
    message(FATAL_ERROR "Missing native LLVM object ${object}")
  endif()
  set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  add_executable(stack_probe_${kib} host.cpp "${object}")
  target_link_libraries(stack_probe_${kib} PRIVATE probe_runtime)
endforeach()
